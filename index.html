async function generateMenu() { const resultArea = document.getElementById("resultArea"); const genBtn = document.getElementById("genBtn"); const people = document.getElementById("peopleSelect").value; const pref = document.getElementById("prefSelect").value; genBtn.disabled = true; resultArea.innerHTML = `
AIが1週間分の最適献立を作成中...
（10〜15秒ほどかかります）
`; const prompt = ` あなたはプロの料理研究家・栄養士です。 以下の条件に合わせて、月曜日から日曜日までの7日分の「朝食」「昼食」「夕食」の献立と簡単なレシピを作成し、指定のJSON形式のみで出力してください。 【条件】 - 人数: ${people} - 食事テーマ: ${pref} - 優先して使いたい食材: ${ingredients.length > 0 ? ingredients.join(", ") : "指定なし"} 【出力フォーマット（必ず以下の厳密なJSON構造のみを出力してください。Markdownなどの装飾は不要）】 { "days": [ { "day": "月曜日", "breakfast": { "name": "料理名", "desc": "一口説明", "ingredients": ["材料1 100g"], "steps": ["手順1"] }, "lunch": { "name": "料理名", "desc": "一口説明", "ingredients": ["材料1"], "steps": ["手順1"] }, "dinner": { "name": "料理名", "desc": "一口説明", "ingredients": ["材料1"], "steps": ["手順1"] } } ] } `; try { // v1beta エンドポイントを使用し、モデル名の指定を修正 const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${API_KEY}`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }], generationConfig: { response_mime_type: "application/json" } }) }); if (!response.ok) { const errData = await response.json(); throw new Error(errData.error?.message || "APIエラー"); } const data = await response.json(); let rawText = data.candidates[0].content.parts[0].text; currentMenuData = JSON.parse(rawText); renderMenu(currentMenuData); } catch (err) { console.error(err); resultArea.innerHTML = `
献立の生成に失敗しました。
詳細: ${err.message}
`; } finally { genBtn.disabled = false; } }
